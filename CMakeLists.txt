cmake_minimum_required(VERSION 3.13)
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(armn_macropad C CXX ASM)
set(CMAKE_C_STANDARD 11)
pico_sdk_init()
if(NOT PICO_BOARD STREQUAL "pico2_w")
  message(FATAL_ERROR "Build with -DPICO_BOARD=pico2_w")
endif()
add_executable(armn_macropad src/main.c src/usb_descriptors.c)
target_include_directories(armn_macropad PRIVATE src)
target_link_libraries(armn_macropad pico_stdlib pico_unique_id tinyusb_device tinyusb_board)
target_compile_options(armn_macropad PRIVATE -Wall -Wextra -Werror)
pico_enable_stdio_usb(armn_macropad 0)
pico_enable_stdio_uart(armn_macropad 0)
pico_set_program_name(armn_macropad "ARMN-Macropad")
pico_set_program_version(armn_macropad "1.0.1")
pico_add_extra_outputs(armn_macropad)
